
Aquí están ambos contratos con todas las vulnerabilidades corregidas, usando el token cUSD en la red Celo.

### 1. Contrato `MessageAccess` (Corregido)

Este contrato ahora utiliza el `ReentrancyGuard` de OpenZeppelin y sigue el patrón Checks-Effects-Interactions para prevenir ataques de reentrancy.

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^^.8.19;

import "@openzeppelin/contracts/security/ReentrancyGuard.sol";
import "@openzeppelin/contracts/token/ERC20/utils/SafeERC20.sol";

/**
 * @title MessageAccess
 * @notice Sistema "paga una vez, mensajea para siempre"
 * @dev Seguro contra reentrancy y sigue el patrón Checks-Effects-Interactions.
 */
interface IERC20 {
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function transfer(address recipient, uint256 amount) external returns (bool);
}

contract MessageAccess is ReentrancyGuard {
    // cUSD en Celo Mainnet
    address public constant TOKEN_ADDRESS = 0x765DE816845861e75A25fCA122bb6898B8B1282a;
    uint256 public constant PLATFORM_FEE = 5; // 5%

    address public owner;
    address public feeCollector;
    bool public paused;

    mapping(address => mapping(address => bool)) public hasPermanentAccess;
    mapping(address => uint256) public accessPrice;

    event AccessPriceSet(address indexed user, uint256 price);
    event AccessGranted(address indexed payer, address indexed receiver, uint256 amount);
    event FeeCollectorUpdated(address indexed newCollector);
    event Paused();
    event Unpaused();

    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;
    }

    modifier whenNotPaused() {
        require(!paused, "Contract paused");
        _;
    }

    constructor(address _feeCollector) {
        require(_feeCollector != address(0), "Invalid collector");
        owner = msg.sender;
        feeCollector = _feeCollector;
    }

    function setAccessPrice(uint256 price) external whenNotPaused {
        accessPrice[msg.sender] = price;
        emit AccessPriceSet(msg.sender, price);
    }

    function payForPermanentAccess(address receiver) external nonReentrant whenNotPaused {
        require(receiver != address(0), "Invalid receiver");
        require(receiver != msg.sender, "Cannot pay yourself");
        require(!hasPermanentAccess[msg.sender][receiver], "Already have access");

        uint256 price = accessPrice[receiver];
        require(price > 0, "Price not set");

        uint256 platformFee = (price * PLATFORM_FEE) / 100;
        uint256 receiverAmount = price - platformFee;

        // Checks-Effects-Interactions pattern
        // 1. Checks (hechas arriba)
        // 2. Effects (actualizar estado ANTES de interacciones externas)
        hasPermanentAccess[msg.sender][receiver] = true;
        
        // 3. Interacciones (llamadas externas)
        IERC20 token = IERC20(TOKEN_ADDRESS);
        require(token.transferFrom(msg.sender, address(this), price), "Transfer to contract failed");
        require(token.transfer(receiver, receiverAmount), "Transfer to receiver failed");
        require(token.transfer(feeCollector, platformFee), "Transfer fee failed");

        emit AccessGranted(msg.sender, receiver, price);
    }

    function canMessage(address sender, address receiver) external view returns (bool) {
        return hasPermanentAccess[sender][receiver];
    }

    function setFeeCollector(address newCollector) external onlyOwner {
        require(newCollector != address(0), "Invalid collector");
        feeCollector = newCollector;
        emit FeeCollectorUpdated(newCollector);
    }

    function pause() external onlyOwner {
        require(!paused, "Already paused");
        paused = true;
        emit Paused();
    }

    function unpause() external onlyOwner {
        require(paused, "Not paused");
        paused = false;
        emit Unpaused();
    }
}
```

### 2. Contrato `GroupAccess` (Corregido)

Este contrato corrige tanto la vulnerabilidad de Inyección de ID de Grupo (usando `keccak256`) como la de reentrancy (usando `ReentrancyGuard` y Checks-Effects-Interactions).

```solidity
// SPDX-License-Identifier: MIT
pragma solidity ^^.8.19;

import "@openzeppelin/contracts/security/ReentrancyGuard.sol";

/**
 * @title GroupAccess
 * @notice Sistema "paga una vez, acceso permanente al grupo"
 * @dev Seguro contra inyección de ID de grupo y reentrancy.
 */
interface IERC20 {
    function transferFrom(address sender, address recipient, uint256 amount) external returns (bool);
    function transfer(address recipient, uint256 amount) external returns (bool);
}

contract GroupAccess is ReentrancyGuard {
    // cUSD en Celo Mainnet
    address public constant TOKEN_ADDRESS = 0x765DE816845861e75A25fCA122bb6898B8B1282a;
    uint256 public constant PLATFORM_FEE = 5; // 5%

    address public owner;
    address public feeCollector;
    bool public paused;

    // Mapeos para grupos (usando bytes32 hash para IDs)
    mapping(bytes32 => address) public groupAdmin;
    mapping(bytes32 => uint256) public groupPrice;
    mapping(address => mapping(bytes32 => bool)) public hasGroupAccess;

    event GroupCreated(string indexed groupId, address indexed admin, uint256 price);
    event GroupAccessGranted(address indexed payer, string indexed groupId, address indexed admin, uint256 amount);
    event FeeCollectorUpdated(address indexed newCollector);
    event Paused();
    event Unpaused();

    modifier onlyOwner() {
        require(msg.sender == owner, "Not owner");
        _;
    }

    modifier whenNotPaused() {
        require(!paused, "Contract paused");
        _;
    }

    constructor(address _feeCollector) {
        require(_feeCollector != address(0), "Invalid collector");
        owner = msg.sender;
        feeCollector = _feeCollector;
    }

    /**
     * @dev Convierte un ID de grupo (string) a un hash bytes32 para usarlo como clave.
     */
    function _getGroupIdHash(string memory groupId) internal pure returns (bytes32) {
        return keccak256(abi.encodePacked(groupId));
    }

    function createPaidGroup(string memory groupId, uint256 price) external whenNotPaused {
        require(bytes(groupId).length > 0, "Invalid group ID");
        require(price > 0, "Price must be > 0");
        
        bytes32 groupIdHash = _getGroupIdHash(groupId);
        
        require(groupAdmin[groupIdHash] == address(0), "Group already exists");
        
        groupAdmin[groupIdHash] = msg.sender;
        groupPrice[groupIdHash] = price;
        hasGroupAccess[msg.sender][groupIdHash] = true;
        
        emit GroupCreated(groupId, msg.sender, price);
    }

    function payForGroupAccess(string memory groupId) external nonReentrant whenNotPaused {
        require(bytes(groupId).length > 0, "Invalid group ID");
        
        bytes32 groupIdHash = _getGroupIdHash(groupId);

        // CHECKS
        require(groupAdmin[groupIdHash] != address(0), "Group does not exist");
        require(groupAdmin[groupIdHash] != msg.sender, "Cannot pay for own group");
        require(!hasGroupAccess[msg.sender][groupIdHash], "Already have access");

        uint256 price = groupPrice[groupIdHash];
        address admin = groupAdmin[groupIdHash];
        uint256 platformFee = (price * PLATFORM_FEE) / 100;
        uint256 adminAmount = price - platformFee;
        IERC20 token = IERC20(TOKEN_ADDRESS);

        // EFFECTS
        hasGroupAccess[msg.sender][groupIdHash] = true;

        // INTERACTIONS
        require(token.transferFrom(msg.sender, address(this), price), "Transfer to contract failed");
        require(token.transfer(admin, adminAmount), "Transfer to admin failed");
        require(token.transfer(feeCollector, platformFee), "Transfer fee failed");

        emit GroupAccessGranted(msg.sender, groupId, admin, price);
    }

    function canAccessGroup(address user, string memory groupId) external view returns (bool) {
        bytes32 groupIdHash = _getGroupIdHash(groupId);
        return hasGroupAccess[user][groupIdHash];
    }

    function getGroupInfo(string memory groupId) external view returns (address admin, uint256 price, bool exists) {
        bytes32 groupIdHash = _getGroupIdHash(groupId);
        admin = groupAdmin[groupIdHash];
        price = groupPrice[groupIdHash];
        exists = admin != address(0);
    }

    function setFeeCollector(address newCollector) external onlyOwner {
        require(newCollector != address(0), "Invalid collector");
        feeCollector = newCollector;
        emit FeeCollectorUpdated(newCollector);
    }

    function pause() external onlyOwner {
        require(!paused, "Already paused");
        paused = true;
        emit Paused();
    }

    function unpause() external onlyOwner {
        require(paused, "Not paused");
        paused = false;
        emit Unpaused();
    }
}
```
